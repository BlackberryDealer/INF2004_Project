# RAM report symbol patterns -- one regex per line.
# Everything after the "#" is the owner column value.
# A symbol is reported for the FIRST pattern it matches.
# Each buddy adds their own symbols here with their own PR (B4.5).
^g_.*_stack$              # B0 integrator (task stacks)
^g_.*_mbf_buf$            # B1/B2/B5 (kernel object buffers)
^g_sb$                    # B5 (status board)
^g_obs_scan$              # B5 (scan table)
^g_echo                   # B5 (echo record and ISR counters)
^g_bar_ring$              # B3 (barcode edge ring)
^g_terrain$               # B4 (terrain monitor words)
^g_speed_hist$            # B4 (speed history words)
^g_imu_prv_[abcdef]$      # B4 (packed imu private words)
^g_ofs_word_[ab]$         # B4 (packed calibration offsets)
^g_ofs_a[xyz]_mg$         # B4 (baseline offsets)
^g_cal_cnt$               # B4 (baseline calibration counter)
^g_run_peak_mm$           # B4 (baseline run peak)
^g_prev_hdg_ddeg$         # B4 (baseline heading reference)
^gb_prev_hdg_valid$       # B4 (baseline heading flag)
^g_prev_pos_[lr]$         # B4 (baseline previous positions)
^g_i2c_err_cnt$           # B4 (baseline I2C error counter)
^g_i2c_good_cnt$          # B4 (baseline good-read counter)
^g_class_gap$             # B4 (baseline class gap)
^g_last_class$            # B4 (baseline last class)
^g_impact_hold$           # B4 (baseline impact hold)
^g_impact_seq_seen$       # B4 (baseline impact sequence)
^g_pitch_ddeg$            # B4 (pitch cache)
^g_hump_now_mm$           # B4 (hump-now cache)
^g_a_fwd_mg$              # B4 (a_fwd cache)
^g_turn_dps$              # B4 (turn-rate cache)
^gb_is_turn_from_enc$     # B4 (turn-source cache)
^g_enc_hdg_prev_cdeg$     # B4 (encoder heading reference)
^g_imu_impact_seq$        # B4 (INT1 sequence byte)
^g_avg_raw_[xyz]$         # B4 (E4 raw accelerometer images)
^g_cal_sum_[xyz]$         # B4 (E2 calibration sums)
^g_mag_m[xyz]$            # B4 (magnetometer cache)
^g_avg_a[xyz]_mg$         # B4 (calibrated average cache)
